// Three-model presentation layer. Keeps the proven simulation engine intact,
// but always renders random, motor-current learning and mapped/RTK together.

floor.geometry.dispose();
floor.geometry=new THREE.PlaneGeometry(120,50);
controls.minDistance=20;
controls.maxDistance=90;

function rebuild(){
  clearSim();
  const poly=SHAPES[shape.value];
  const b=bounds(poly), lawnWidth=b.maxX-b.minX;
  const sep=lawnWidth+2.5;
  worlds=[
    createWorld(poly,-sep,'random'),
    createWorld(poly,0,'adaptive'),
    createWorld(poly,sep,'smart')
  ];

  // Frame all three lawns automatically when shape changes.
  const totalSpan=2*sep+lawnWidth;
  camera.position.set(0,Math.max(22,totalSpan*.42),Math.max(38,totalSpan*1.02));
  controls.target.set(0,0,0);
  controls.update();
  mc.textContent='';
  updateStats();
}

function updateStats(){
  const r=worlds[0],a=worlds[1],s=worlds[2];
  if(!r||!a||!s)return;
  const pct=x=>(100*x).toFixed(1).replace('.',',')+' %';
  const A=areaPoly(r.poly);
  const eff=w=>w.distance?Math.min(1,A*coverage(w)/(w.distance*cutWidth)):0;
  const el=id=>document.getElementById(id);

  el('rCov').textContent=pct(coverage(r));
  el('aCov').textContent=pct(coverage(a));
  el('sCov').textContent=pct(coverage(s));
  el('rDist').textContent=r.distance.toFixed(1).replace('.',',')+' m';
  el('aDist').textContent=a.distance.toFixed(1).replace('.',',')+' m';
  el('sDist').textContent=s.distance.toFixed(1).replace('.',',')+' m';
  el('rTime').textContent=fmtTime(r.elapsed);
  el('aTime').textContent=fmtTime(a.elapsed);
  el('sTime').textContent=fmtTime(s.elapsed);

  for(const [w,barId,txtId] of [[r,'rBat','rBatTxt'],[a,'aBat','aBatTxt'],[s,'sBat','sBatTxt']]){
    const bar=el(barId),txt=el(txtId);
    bar.style.width=(100*w.battery)+'%';
    txt.textContent=Math.round(100*w.battery)+' %';
    bar.style.background=w.battery<.2?'var(--danger)':'var(--accent)';
  }

  el('aMotor').textContent=`${a.motorCurrent.toFixed(1)} A · ${a.learn?Array.from(a.learn.visits).filter(Boolean).length:0} lärda celler`;
  el('rExtra').textContent=`unik/svept ${Math.round(100*eff(r))}% · laddningar ${r.chargeCount}`;
  el('aExtra').textContent=`unik/svept ${Math.round(100*eff(a))}% · laddningar ${a.chargeCount}`;
  el('sExtra').textContent=s.state==='blocked'
    ? 'STOPP: säkerhetsgräns träffad'
    : `unik/svept ${Math.round(100*eff(s))}% · laddningar ${s.chargeCount}`;
}
