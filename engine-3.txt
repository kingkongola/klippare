  const w={poly,offset,kind,lawn,grass,mower,dockPos:dp,x:start[0],y:start[1],heading:Math.random()*Math.PI*2,speed:kind==='smart'?.48:.43,turnRate:kind==='smart'?1.55:1.35,
    bodyR:MOWER_CLEARANCE,distance:0,elapsed:0,energy:0,batteryWh:kind==='smart'?82:70,battery:1,chargeCount:0,state:'work',reverseLeft:0,turnLeft:0,
    route,routeIndex:0,routeMow:true,returnRoute:[],returnIndex:0,resume:null,covered:0,texTick:0,done:false,motorCurrent:1.25,learn:null};
  if(kind==='adaptive')w.learn=makeLearningMap(poly,start);
  updateMower(w);return w
}
function updateMower(w){w.mower.position.set(w.offset+w.x,.02,w.y);w.mower.rotation.y=-w.heading}
function useEnergy(w,power,dt){w.energy+=power*dt/3600;w.battery=Math.max(0,w.battery-power*dt/3600/w.batteryWh)}
function markCut(w,dt,active=true){
  if(active){
    w.motorCurrent=sampleMotorCurrent(w);
    if(w.kind==='adaptive')adaptiveObserve(w,w.motorCurrent);
    w.covered+=w.lawn.mark(w.x,w.y,cutWidth/2);
    cutGrassField(w.grass,w.x,w.y,cutWidth/2);
    w.texTick+=dt;if(w.texTick>.15){w.lawn.draw();w.texTick=0}
  }
}
function startReturn(w){
  w.resume={x:w.x,y:w.y,heading:w.heading,state:w.state,routeIndex:w.routeIndex};
  const route=bfsPath([w.x,w.y],w.dockPos,w.poly,.28,w.bodyR);
  if(!route){w.state='blocked';return false}
  w.returnRoute=route;w.returnIndex=0;w.state='return';return true
}
function followPoints(w,points,indexKey,dt,mow=false){
  if(w[indexKey]>=points.length)return true;
  const target=points[w[indexKey]],p=target.p||target,dx=p[0]-w.x,dy=p[1]-w.y,d=Math.hypot(dx,dy),desired=Math.atan2(dy,dx),da=angDiff(w.heading,desired);
  if(Math.abs(da)>.10){const step=Math.sign(da)*Math.min(Math.abs(da),w.turnRate*dt);w.heading+=step;useEnergy(w,w.kind==='smart'?42:34,dt);return false}
  const move=Math.min(d,w.speed*dt);
  if(d>1e-5){
    const nx=w.x+dx/d*move,ny=w.y+dy/d*move;
    if(!safeAt(nx,ny,w.poly,w.bodyR) || !straightInside([w.x,w.y],[nx,ny],w.poly,w.bodyR)){
      w.state='blocked';return false;
    }
    w.x=nx;w.y=ny;w.distance+=move;
    useEnergy(w,w.kind==='smart'?64:52,dt);
    markCut(w,dt,mow&&(target.mow??true));
  }
  if(d<.055||move>=d-.001)w[indexKey]++;
  return w[indexKey]>=points.length
}
function stepRandom(w,dt){
  if(w.state==='charge'){w.elapsed+=dt;w.battery=Math.min(1,w.battery+120*dt/3600/w.batteryWh);if(w.battery>=.96){w.chargeCount++;w.state='resume';w.returnRoute=bfsPath(w.dockPos,[w.resume.x,w.resume.y],w.poly,.28,w.bodyR)||[];w.returnIndex=0}return}
  if(w.state==='return'||w.state==='resume'){const done=followPoints(w,w.returnRoute,'returnIndex',dt,false);w.elapsed+=dt;if(done){if(w.state==='return'){w.state='charge';w.battery=Math.max(w.battery,.01)}else{w.state='drive';w.heading=w.resume.heading}}return}
  if(w.battery<.14){startReturn(w);return}
  w.elapsed+=dt;
  if(w.state==='reverse'){const mv=Math.min(.18*dt,w.reverseLeft);const nx=w.x-Math.cos(w.heading)*mv,ny=w.y-Math.sin(w.heading)*mv;if(safeAt(nx,ny,w.poly,w.bodyR)){const dx=nx-w.x,dy=ny-w.y;w.x=nx;w.y=ny;updateOdometry(w,dx,dy);w.distance+=mv;w.reverseLeft-=mv;useEnergy(w,44,dt)}else w.reverseLeft=0;if(w.reverseLeft<=0){if(w.kind==='adaptive')w.turnLeft=chooseAdaptiveTurn(w);w.state='turn'}return}
  if(w.state==='turn'){const da=Math.sign(w.turnLeft)*Math.min(Math.abs(w.turnLeft),w.turnRate*dt);w.heading+=da;w.turnLeft-=da;useEnergy(w,36,dt);if(Math.abs(w.turnLeft)<.02)w.state='drive';return}
  w.heading+=(Math.random()-.5)*.012*dt;
  const mv=w.speed*dt,nx=w.x+Math.cos(w.heading)*mv,ny=w.y+Math.sin(w.heading)*mv;
  if(!safeAt(nx,ny,w.poly,w.bodyR)){w.state='reverse';w.reverseLeft=.18+.12*Math.random();w.turnLeft=w.kind==='adaptive'?0:(Math.random()<.5?-1:1)*(1.15+Math.random()*1.55);return}
  const dx=nx-w.x,dy=ny-w.y;w.x=nx;w.y=ny;updateOdometry(w,dx,dy);w.distance+=mv;useEnergy(w,52,dt);markCut(w,dt,true)
}
function stepSmart(w,dt){
  if(w.done||w.state==='blocked')return;
  if(w.state==='charge'){w.elapsed+=dt;w.battery=Math.min(1,w.battery+135*dt/3600/w.batteryWh);if(w.battery>=.96){w.chargeCount++;w.state='resume';w.returnRoute=bfsPath(w.dockPos,[w.resume.x,w.resume.y],w.poly,.28,w.bodyR)||[];w.returnIndex=0}return}
  if(w.state==='return'||w.state==='resume'){const done=followPoints(w,w.returnRoute,'returnIndex',dt,false);w.elapsed+=dt;if(done){if(w.state==='return')w.state='charge';else{w.state='work';w.routeIndex=w.resume.routeIndex}}return}
  if(w.battery<.14){startReturn(w);return}
  w.elapsed+=dt;
  if(w.routeIndex>=w.route.length){w.done=true;return}
  const target=w.route[w.routeIndex],dx=target.p[0]-w.x,dy=target.p[1]-w.y,d=Math.hypot(dx,dy),desired=Math.atan2(dy,dx),da=angDiff(w.heading,desired);
  if(Math.abs(da)>.08){const step=Math.sign(da)*Math.min(Math.abs(da),w.turnRate*dt);w.heading+=step;useEnergy(w,43,dt);return}
  const mv=Math.min(d,w.speed*dt);
  if(d>1e-5){
    const nx=w.x+dx/d*mv,ny=w.y+dy/d*mv;
    if(!safeAt(nx,ny,w.poly,w.bodyR) || !straightInside([w.x,w.y],[nx,ny],w.poly,w.bodyR)){
      w.state='blocked';return;
    }
    w.x=nx;w.y=ny;w.distance+=mv;useEnergy(w,64,dt);markCut(w,dt,target.mow);
  }
  if(d<.05||mv>=d-.001)w.routeIndex++
}

async function monteCarlo(trials=150){
  mc.textContent='Kör Monte Carlo…';await new Promise(r=>setTimeout(r,20));
  const poly=SHAPES[shape.value],b=bounds(poly),N=76,H=Math.max(30,Math.round(N*(b.maxY-b.minY)/(b.maxX-b.minX))),W=N,dx=(b.maxX-b.minX)/W,dy=(b.maxY-b.minY)/H,cells=[],cellAt=new Int32Array(W*H);cellAt.fill(-1);
  for(let iy=0;iy<H;iy++)for(let ix=0;ix<W;ix++){const x=b.minX+(ix+.5)*dx,y=b.minY+(iy+.5)*dy;if(pointInPoly(x,y,poly)){cellAt[iy*W+ix]=cells.length;cells.push([x,y])}}
  const need=Math.ceil(.95*cells.length),res=[],ds=.18,maxDist=areaPoly(poly)/cutWidth*12,r=cutWidth/2,r2=r*r;
  for(let t=0;t<trials;t++){
    let [x,y]=randomPoint(poly,.46),heading=Math.random()*Math.PI*2,dist=0,count=0,state='drive',back=0,turn=0;
    const cut=new Uint8Array(cells.length);
    while(count<need&&dist<maxDist){
      if(state==='reverse'){const mv=Math.min(ds*.45,back),nx=x-Math.cos(heading)*mv,ny=y-Math.sin(heading)*mv;if(safeAt(nx,ny,poly,.46)){x=nx;y=ny;dist+=mv}back-=mv;if(back<=0)state='turn'}
      else if(state==='turn'){const da=Math.sign(turn)*Math.min(Math.abs(turn),.33);heading+=da;turn-=da;if(Math.abs(turn)<.02)state='drive'}
      else{const nx=x+Math.cos(heading)*ds,ny=y+Math.sin(heading)*ds;if(!safeAt(nx,ny,poly,.46)){state='reverse';back=.18+.12*Math.random();turn=(Math.random()<.5?-1:1)*(1.15+Math.random()*1.55)}else{x=nx;y=ny;dist+=ds}}
      const cx=Math.floor((x-b.minX)/dx),cy=Math.floor((y-b.minY)/dy),rx=Math.ceil(r/dx),ry=Math.ceil(r/dy);
      for(let gy=Math.max(0,cy-ry);gy<=Math.min(H-1,cy+ry);gy++)for(let gx=Math.max(0,cx-rx);gx<=Math.min(W-1,cx+rx);gx++){const ci=cellAt[gy*W+gx];if(ci<0||cut[ci])continue;const ddx=cells[ci][0]-x,ddy=cells[ci][1]-y;if(ddx*ddx+ddy*ddy<=r2){cut[ci]=1;count++}}
    }
    res.push(dist);if(t%15===0){mc.textContent=`Kör Monte Carlo… ${t}/${trials}`;await new Promise(r=>setTimeout(r,0))}
  }
  res.sort((a,b)=>a-b);const mean=res.reduce((a,b)=>a+b,0)/res.length,q=p=>res[Math.floor((res.length-1)*p)];
  const route=smartRoute(poly,cutWidth);let smart=0;let prev=route[0]?.p;for(let i=1;i<route.length;i++){smart+=Math.hypot(route[i].p[0]-prev[0],route[i].p[1]-prev[1]);prev=route[i].p}
  mc.innerHTML=`<strong>95 % täckning</strong><br>Slump: medel ${mean.toFixed(0)} m, median ${q(.5).toFixed(0)} m, 10–90 % ${q(.1).toFixed(0)}–${q(.9).toFixed(0)} m.<br>Smart planerad helrutt: ca ${smart.toFixed(0)} m. Sträckförhållande ≈ <strong>${(mean/smart).toFixed(1)}×</strong>.`;
}
