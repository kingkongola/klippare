// Honest dead-reckoning experiment.
// Models 2 and 3 use only their own wheel-derived pose estimate, their sparse
// learned map, and boundary contacts already experienced. The motor variant
// additionally gets local blade-current observations. Neither may query the
// true lawn polygon while choosing a turn.

const DR_TURN_CANDIDATES=[-.55,.55,-.9,.9,-1.25,1.25,-1.62,1.62,-2.05,2.05,-2.55,2.55];

makeLearningMap = function(_poly,start){
  return Core.createEstimator({
    x:start[0], y:start[1], heading:0,
    profile:Core.makeProfile({driftPct:2,seed:1,wheelBase:.48}),
    useMotor:false
  });
};

function configureEstimator(w,{start,heading,profile,useMotor}){
  w.x=start[0];w.y=start[1];w.heading=heading;
  w.useMotor=useMotor;
  // Dead reckoning gets an arbitrary local origin, not the true world
  // coordinate or an absolute compass heading.
  w.drOrigin={x:start[0],y:start[1],heading};
  w.learn=Core.createEstimator({
    x:0,y:0,heading:0,
    profile:{...profile},
    useMotor
  });
  updateMower(w);
}

function grassSignalMode(){
  return document.getElementById('grassMode')?.value||'patchy';
}

grassDensity = function(x,y){
  const mode=grassSignalMode();
  if(mode==='uniform'||mode==='short')return .96;
  return Math.max(.65,Math.min(1.25,.91+.13*Math.sin(x*.73+y*.27)+.10*Math.cos(y*.59-x*.31)));
};

sampleMotorCurrent = function(w){
  const fresh=w.lawn.sampleUncut(w.x,w.y,cutWidth*.48);
  const density=grassDensity(w.x,w.y);
  const mode=grassSignalMode();
  const heightFactor=mode==='short'?.16:1;
  // Local physical measurement only: true grass under the blade is allowed.
  // No information about future cells is exposed.
  const noise=(Math.random()-.5)*(mode==='short'?.22:.16);
  return Math.max(1.05,1.25+3.8*fresh*density*heightFactor+noise);
};

adaptiveObserve = function(w,current){
  if(!w.learn)return;
  const norm=Math.max(0,Math.min(1,(current-1.25)/4.0));
  Core.observePass(w.learn,w.useMotor?norm:0);
};

updateOdometry = function(w,dx,dy){
  if(!w.learn)return;
  // Project actual displacement onto the robot's true forward axis to obtain
  // the wheel travel that encoders would observe. The estimator never receives
  // true x/y.
  const ds=dx*Math.cos(w.heading)+dy*Math.sin(w.heading);
  Core.integrateTrueWheelMotion(w.learn,ds,ds);
};

function updateOdometryTurn(w,dtheta){
  if(!w.learn)return;
  const b=w.learn.profile.wheelBase;
  Core.integrateTrueWheelMotion(w.learn,-b*dtheta/2,b*dtheta/2);
}

function observeLearnedBoundary(w){
  if(w.learn)Core.observeBoundary(w.learn,w.bodyR*.92);
}

adaptiveRayScore = function(w,relativeAngle){
  if(!w.learn)return 0;
  return Core.scoreRay(w.learn,relativeAngle,{maxDistance:4.2,step:.36,motorWeight:.55});
};

chooseAdaptiveTurn = function(w){
  if(!w.learn)return(Math.random()<.5?-1:1)*(1.05+Math.random()*1.8);
  return Core.chooseTurn(w.learn,DR_TURN_CANDIDATES,{
    exploration:.14,
    random:w.learn.rng,
    scoreOptions:{maxDistance:4.2,step:.36,motorWeight:.55}
  });
};

markCut = function(w,dt,active=true){
  if(active){
    w.motorCurrent=sampleMotorCurrent(w);
    if(w.kind==='adaptive')adaptiveObserve(w,w.motorCurrent);
    w.covered+=w.lawn.mark(w.x,w.y,cutWidth/2);
    cutGrassField(w.grass,w.x,w.y,cutWidth/2);
    w.texTick+=dt;
    if(w.texTick>.15){w.lawn.draw();w.texTick=0}
  }
};

stepRandom = function(w,dt){
  if(w.state==='charge'){
    w.elapsed+=dt;
    w.battery=Math.min(1,w.battery+120*dt/3600/w.batteryWh);
    if(w.battery>=.96){
      w.chargeCount++;w.state='resume';
      w.returnRoute=bfsPath(w.dockPos,[w.resume.x,w.resume.y],w.poly,.28,w.bodyR)||[];
      w.returnIndex=0
    }
    return
  }
  if(w.state==='return'||w.state==='resume'){
    const done=followPoints(w,w.returnRoute,'returnIndex',dt,false);
    w.elapsed+=dt;
    if(done){
      if(w.state==='return'){w.state='charge';w.battery=Math.max(w.battery,.01)}
      else{w.state='drive';w.heading=w.resume.heading}
    }
    return
  }
  if(w.battery<.14){startReturn(w);return}
  w.elapsed+=dt;

  if(w.state==='reverse'){
    const mv=Math.min(.18*dt,w.reverseLeft);
    const nx=w.x-Math.cos(w.heading)*mv,ny=w.y-Math.sin(w.heading)*mv;
    if(safeAt(nx,ny,w.poly,w.bodyR)){
      const dx=nx-w.x,dy=ny-w.y;
      w.x=nx;w.y=ny;
      updateOdometry(w,dx,dy);
      w.distance+=mv;w.reverseLeft-=mv;useEnergy(w,44,dt)
    }else w.reverseLeft=0;
    if(w.reverseLeft<=0){
      if(w.kind==='adaptive')w.turnLeft=chooseAdaptiveTurn(w);
      w.state='turn'
    }
    return
  }

  if(w.state==='turn'){
    const da=Math.sign(w.turnLeft)*Math.min(Math.abs(w.turnLeft),w.turnRate*dt);
    w.heading+=da;w.turnLeft-=da;
    updateOdometryTurn(w,da);
    useEnergy(w,36,dt);
    if(Math.abs(w.turnLeft)<.02)w.state='drive';
    return
  }

  // Small real-world heading disturbance. It affects the actual robot but is
  // not handed directly to dead reckoning.
  w.heading+=(Math.random()-.5)*.012*dt;

  const mv=w.speed*dt,nx=w.x+Math.cos(w.heading)*mv,ny=w.y+Math.sin(w.heading)*mv;
  // safeAt is used here only as the physical bumper/boundary event for the
  // next movement increment. It is NOT used by the turn planner.
  if(!safeAt(nx,ny,w.poly,w.bodyR)){
    if(w.kind==='adaptive')observeLearnedBoundary(w);
    w.state='reverse';
    w.reverseLeft=.18+.12*Math.random();
    w.turnLeft=w.kind==='adaptive'?0:(Math.random()<.5?-1:1)*(1.15+Math.random()*1.55);
    return
  }

  const dx=nx-w.x,dy=ny-w.y;
  w.x=nx;w.y=ny;
  updateOdometry(w,dx,dy);
  w.distance+=mv;useEnergy(w,52,dt);markCut(w,dt,true)
};

function addModelMarker(w,color){
  const mat=new THREE.MeshStandardMaterial({color,emissive:color,emissiveIntensity:.18,roughness:.55});
  const puck=new THREE.Mesh(new THREE.CylinderGeometry(.13,.13,.09,20),mat);
  puck.position.set(0,.61,0);
  w.mower.add(puck);
}

function estimatorError(w){
  if(!w.learn||!w.drOrigin)return 0;
  const dx=w.x-w.drOrigin.x,dy=w.y-w.drOrigin.y,h=w.drOrigin.heading;
  // Express the true displacement in the estimator's start-relative frame.
  const tx=Math.cos(h)*dx+Math.sin(h)*dy;
  const ty=-Math.sin(h)*dx+Math.cos(h)*dy;
  return Core.positionError(w.learn,tx,ty);
}

rebuild = function(){
  clearSim();
  const poly=SHAPES[shape.value];
  const b=bounds(poly),lawnWidth=b.maxX-b.minX;
  const sep=lawnWidth+2.15;
  const offsets=[-1.5*sep,-.5*sep,.5*sep,1.5*sep];

  const r=createWorld(poly,offsets[0],'random');
  const mem=createWorld(poly,offsets[1],'adaptive');
  const motor=createWorld(poly,offsets[2],'adaptive');
  const smart=createWorld(poly,offsets[3],'smart');

  // Sensor ablation: models 2 and 3 begin with exactly the same true pose and
  // exactly the same odometry calibration/noise seed. Their only intentional
  // algorithmic difference is whether blade current may influence the map.
  const start=randomPoint(poly,MOWER_CLEARANCE);
  const heading=Math.random()*Math.PI*2;
  const drift=+(document.getElementById('drift')?.value||2);
  const seed=(Math.random()*0xffffffff)>>>0;
  const profile=Core.makeProfile({driftPct:drift,seed,wheelBase:.48});
  configureEstimator(mem,{start,heading,profile,useMotor:false});
  configureEstimator(motor,{start,heading,profile,useMotor:true});

  mem.variant='memory';
  motor.variant='motor';
  addModelMarker(mem,0xd7b356);
  addModelMarker(motor,0x59a6d8);

  worlds=[r,mem,motor,smart];
  fitCameraToLawns(poly,offsets);
  mc.textContent='';
  updateStats();
};

updateStats = function(){
  const r=worlds[0],m=worlds[1],a=worlds[2],s=worlds[3];
  if(!r||!m||!a||!s)return;
  const pct=x=>(100*x).toFixed(1).replace('.',',')+' %';
  const A=areaPoly(r.poly);
  const eff=w=>w.distance?Math.min(1,A*coverage(w)/(w.distance*cutWidth)):0;
  const el=id=>document.getElementById(id);
  const fmt=x=>x.toFixed(1).replace('.',',');

  for(const [w,prefix] of [[r,'r'],[m,'m'],[a,'a'],[s,'s']]){
    el(prefix+'Cov').textContent=pct(coverage(w));
    el(prefix+'Dist').textContent=fmt(w.distance)+' m';
    el(prefix+'Time').textContent=fmtTime(w.elapsed);
    const bar=el(prefix+'Bat'),txt=el(prefix+'BatTxt');
    bar.style.width=(100*w.battery)+'%';
    txt.textContent=Math.round(100*w.battery)+' %';
    bar.style.background=w.battery<.2?'var(--danger)':'var(--accent)';
  }

  el('rExtra').textContent=`unik/svept ${Math.round(100*eff(r))}% · laddningar ${r.chargeCount}`;
  el('mExtra').textContent=`kartceller ${m.learn?.cells.size||0} · positionsfel ${fmt(estimatorError(m))} m`;
  el('aMotor').textContent=`${fmt(a.motorCurrent)} A`;
  el('aExtra').textContent=`kartceller ${a.learn?.cells.size||0} · positionsfel ${fmt(estimatorError(a))} m`;
  el('sExtra').textContent=s.state==='blocked'
    ? 'STOPP: säkerhetsgräns träffad'
    : `unik/svept ${Math.round(100*eff(s))}% · laddningar ${s.chargeCount}`;
};

monteCarlo = async function(trials=100){
  mc.innerHTML='<div class="mc-summary"><strong>Kör 100 jämförbara försök…</strong><br>Varje försök startar med en ny, oklippt gräsmatta. Vi mäter meter tills 95 % är klippt.</div>';
  await new Promise(r=>setTimeout(r,20));

  const poly=SHAPES[shape.value],b=bounds(poly);
  const N=74,W=N,H=Math.max(30,Math.round(N*(b.maxY-b.minY)/(b.maxX-b.minX)));
  const dx=(b.maxX-b.minX)/W,dy=(b.maxY-b.minY)/H;
  const cells=[],cellAt=new Int32Array(W*H);cellAt.fill(-1);
  for(let iy=0;iy<H;iy++)for(let ix=0;ix<W;ix++){
    const x=b.minX+(ix+.5)*dx,y=b.minY+(iy+.5)*dy;
    if(pointInPoly(x,y,poly)){cellAt[iy*W+ix]=cells.length;cells.push([x,y])}
  }
  const safeStarts=cells.filter(([x,y])=>safeAt(x,y,poly,.46));
  const need=Math.ceil(.95*cells.length),radius=cutWidth/2,r2=radius*radius;
  const maxDist=areaPoly(poly)/cutWidth*15;
  const drift=+(document.getElementById('drift')?.value||2);
  const mode=grassSignalMode();

  function visitAround(cut,x,y,doCut=true){
    const cx=Math.floor((x-b.minX)/dx),cy=Math.floor((y-b.minY)/dy);
    const rx=Math.ceil(radius/dx),ry=Math.ceil(radius/dy);
    let added=0,total=0,uncut=0;
    for(let gy=Math.max(0,cy-ry);gy<=Math.min(H-1,cy+ry);gy++)for(let gx=Math.max(0,cx-rx);gx<=Math.min(W-1,cx+rx);gx++){
      const ci=cellAt[gy*W+gx];if(ci<0)continue;
      const ddx=cells[ci][0]-x,ddy=cells[ci][1]-y;
      if(ddx*ddx+ddy*ddy>r2)continue;
      total++;if(!cut[ci])uncut++;
      if(doCut&&!cut[ci]){cut[ci]=1;added++}
    }
    return{added,fresh:total?uncut/total:0};
  }

  function currentAt(cut,x,y,rng){
    const fresh=visitAround(cut,x,y,false).fresh;
    const density=mode==='patchy'
      ? Math.max(.65,Math.min(1.25,.91+.13*Math.sin(x*.73+y*.27)+.10*Math.cos(y*.59-x*.31)))
      : .96;
    const height=mode==='short'?.16:1;
    const noise=(rng()-.5)*(mode==='short'?.22:.16);
    return Math.max(1.05,1.25+3.8*fresh*density*height+noise);
  }

  function reactive(kind,start,heading,seed,profile){
    const rng=Core.mulberry32(seed);
    const cut=new Uint8Array(cells.length);
    let x=start[0],y=start[1],h=heading,dist=0,count=0;
    const useMemory=kind!=='random',useMotor=kind==='motor';
    const est=useMemory?Core.createEstimator({x:0,y:0,heading:0,profile:{...profile},useMotor}):null;

    while(count<need&&dist<maxDist){
      // Small unobserved physical heading disturbance.
      h+=(rng()-.5)*.004;
      const step=.18,nx=x+Math.cos(h)*step,ny=y+Math.sin(h)*step;
      // True geometry is consulted only for the next physical movement: this
      // represents actually reaching the boundary, never lookahead planning.
      if(!safeAt(nx,ny,poly,.46)){
        if(est)Core.observeBoundary(est,.46);
        const back=.24,bx=x-Math.cos(h)*back,by=y-Math.sin(h)*back;
        if(safeAt(bx,by,poly,.46)){
          x=bx;y=by;dist+=back;
          if(est)Core.integrateTrueWheelMotion(est,-back,-back);
        }
        let turn;
        if(est){
          turn=Core.chooseTurn(est,DR_TURN_CANDIDATES,{
            exploration:.14,random:est.rng,
            scoreOptions:{maxDistance:4.2,step:.36,motorWeight:.55}
          });
          const wb=est.profile.wheelBase;
          Core.integrateTrueWheelMotion(est,-wb*turn/2,wb*turn/2);
        }else{
          turn=(rng()<.5?-1:1)*(1.15+rng()*1.55);
        }
        h+=turn;
        continue;
      }
      x=nx;y=ny;dist+=step;
      if(est)Core.integrateTrueWheelMotion(est,step,step);

      // Both DR variants take the same local physical sample so their random
      // streams stay paired. Only the motor variant is allowed to use it.
      const amps=currentAt(cut,x,y,rng);
      const v=visitAround(cut,x,y,true);count+=v.added;
      if(est)Core.observePass(est,useMotor?Math.max(0,Math.min(1,(amps-1.25)/4)):0);
    }
    return{distance:dist,reached:count>=need};
  }

  function planned95(start){
    const mowing=smartRoute(poly,cutWidth,start);
    if(!mowing.length)return{distance:maxDist,reached:false};
    const route=[];
    const outbound=bfsPath(start,mowing[0].p,poly,.28,MOWER_CLEARANCE);
    if(outbound)for(const q of outbound)route.push({p:q,mow:false});
    route.push(...mowing);
    const cut=new Uint8Array(cells.length);
    let prev=start,dist=0,count=0;
    for(const target of route){
      const p=target.p,seg=Math.hypot(p[0]-prev[0],p[1]-prev[1]);
      const n=Math.max(1,Math.ceil(seg/.14));
      for(let i=1;i<=n;i++){
        const t=i/n,x=prev[0]+(p[0]-prev[0])*t,y=prev[1]+(p[1]-prev[1])*t;
        dist+=seg/n;
        if(target.mow){const v=visitAround(cut,x,y,true);count+=v.added}
        if(count>=need)return{distance:dist,reached:true};
      }
      prev=p;
    }
    return{distance:dist,reached:count>=need};
  }

  const runs=[];
  for(let t=0;t<trials;t++){
    const baseSeed=(0x9e3779b9^(t*2654435761))>>>0;
    const rng=Core.mulberry32(baseSeed);
    const start=safeStarts[Math.floor(rng()*safeStarts.length)];
    const heading=rng()*Math.PI*2;
    const profile=Core.makeProfile({driftPct:drift,seed:(baseSeed^0xa5a5a5a5)>>>0,wheelBase:.48});
    runs.push({
      random:reactive('random',start,heading,(baseSeed^0x11111111)>>>0,profile),
      memory:reactive('memory',start,heading,(baseSeed^0x22222222)>>>0,profile),
      motor:reactive('motor',start,heading,(baseSeed^0x22222222)>>>0,profile),
      smart:planned95(start)
    });
    if((t+1)%5===0||t===0){
      mc.innerHTML=`<div class="mc-summary"><strong>Kör försök ${t+1} av ${trials}…</strong><br>Varje robot stoppas när 95 % av gräset är klippt.</div>`;
      await new Promise(r=>setTimeout(r,0));
    }
  }

  function quantile(xs,p){
    if(!xs.length)return NaN;
    const i=Math.max(0,Math.min(xs.length-1,Math.floor((xs.length-1)*p)));
    return xs[i];
  }
  function summary(key){
    const vals=runs.map(r=>r[key]);
    const success=vals.filter(v=>v.reached).map(v=>v.distance).sort((a,b)=>a-b);
    return{
      success:success.length,
      rate:success.length/trials,
      med:quantile(success,.5),
      q10:quantile(success,.1),
      q90:quantile(success,.9)
    };
  }
  const R=summary('random'),M=summary('memory'),A=summary('motor'),S=summary('smart');

  const fmt=v=>Number.isFinite(v)?`${Math.round(v)} m`:'–';
  const range=x=>Number.isFinite(x.q10)?`${Math.round(x.q10)}–${Math.round(x.q90)} m`:'–';
  const success=x=>`${x.success}/${trials}`;
  const pctShorter=(base,x)=>Number.isFinite(base.med)&&Number.isFinite(x.med)?100*(base.med-x.med)/base.med:NaN;
  const vsRandom=x=>{
    const g=pctShorter(R,x);
    if(!Number.isFinite(g))return '–';
    return g>=0?`${Math.round(g)} % kortare`:`${Math.round(-g)} % längre`;
  };
  const motorGain=pctShorter(M,A);
  const memGain=pctShorter(R,M);

  let takeaway='';
  if(Number.isFinite(memGain)&&Number.isFinite(motorGain)){
    const memMeters=R.med-M.med;
    const motorMeters=M.med-A.med;
    takeaway=`<strong>Så läser du resultatet:</strong> minnet ändrar typkörningen med ${Math.abs(Math.round(memMeters))} m (${Math.abs(Math.round(memGain))} %) jämfört med ren slump. `;
    if(motorGain>2){
      takeaway+=`Klippmotorströmmen sparar sedan ytterligare <strong>${Math.round(motorMeters)} m (${Math.round(motorGain)} %)</strong> jämfört med minne utan motorström.`;
    }else if(motorGain<-2){
      takeaway+=`I just denna inställning gör klippmotorströmmen faktiskt modellen <strong>${Math.round(-motorGain)} % sämre</strong> än minne utan motorström.`;
    }else{
      takeaway+=`Klippmotorströmmen gör <strong>nästan ingen mätbar skillnad</strong> utöver minnet i just denna inställning.`;
    }
  }

  mc.innerHTML=`
    <p class="mc-summary"><strong>${trials} oberoende försök, mål: 95 % klippt.</strong><br>
    Siffran “typiskt” är medianen: hälften av körningarna var kortare och hälften längre. Lägre är bättre.</p>
    <table class="mc-table">
      <thead><tr><th>Strategi</th><th>Typiskt till 95 %</th><th>Jämfört med slump</th><th>80 % av körningarna</th><th>Nådde 95 %</th></tr></thead>
      <tbody>
        <tr><td>1. Ren slump</td><td><strong>${fmt(R.med)}</strong></td><td>referens</td><td>${range(R)}</td><td>${success(R)}</td></tr>
        <tr><td>2. DR + minne</td><td><strong>${fmt(M.med)}</strong></td><td class="${pctShorter(R,M)>0?'mc-win':''}">${vsRandom(M)}</td><td>${range(M)}</td><td>${success(M)}</td></tr>
        <tr><td>3. + klippmotor</td><td><strong>${fmt(A.med)}</strong></td><td class="${pctShorter(R,A)>0?'mc-win':''}">${vsRandom(A)}</td><td>${range(A)}</td><td>${success(A)}</td></tr>
        <tr><td>4. Kartlagd / RTK</td><td><strong>${fmt(S.med)}</strong></td><td class="${pctShorter(R,S)>0?'mc-win':''}">${vsRandom(S)}</td><td>${range(S)}</td><td>${success(S)}</td></tr>
      </tbody>
    </table>
    <p class="mc-summary">${takeaway}</p>
    <div class="mc-note">“80 % av körningarna” betyder intervallet från 10:e till 90:e percentilen. Om en strategi inte når 95 % inom ${Math.round(maxDist)} m räknas den som misslyckad i kolumnen längst till höger, inte som ett falskt fint resultat. Modell 2 och 3 får samma start och samma DR-profil; deras avsiktliga skillnad är bara klippmotor-signalen.</div>`;
};
