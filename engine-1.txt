function bounds(p){const xs=p.map(x=>x[0]),ys=p.map(x=>x[1]);return{minX:Math.min(...xs),maxX:Math.max(...xs),minY:Math.min(...ys),maxY:Math.max(...ys)}}
function pointInPoly(x,y,p){let inside=false;for(let i=0,j=p.length-1;i<p.length;j=i++){const[xi,yi]=p[i],[xj,yj]=p[j];if(((yi>y)!=(yj>y))&&x<(xj-xi)*(y-yi)/(yj-yi)+xi)inside=!inside}return inside}
const MOWER_CLEARANCE=.74;

function pointSegmentDistance(px,py,ax,ay,bx,by){
  const vx=bx-ax,vy=by-ay,wx=px-ax,wy=py-ay;
  const vv=vx*vx+vy*vy;
  const t=vv?Math.max(0,Math.min(1,(wx*vx+wy*vy)/vv)):0;
  const dx=px-(ax+t*vx),dy=py-(ay+t*vy);
  return Math.hypot(dx,dy);
}
function safeAt(x,y,p,r=MOWER_CLEARANCE){
  if(!pointInPoly(x,y,p))return false;
  // Exact circular-footprint constraint: the mower centre must remain at
  // least r metres from every boundary segment. For this simple polygon
  // that means the whole robot disc is guaranteed to stay on grass.
  for(let i=0,j=p.length-1;i<p.length;j=i++){
    if(pointSegmentDistance(x,y,p[j][0],p[j][1],p[i][0],p[i][1])<r)return false;
  }
  return true;
}
function randomPoint(poly,r=MOWER_CLEARANCE){const b=bounds(poly);for(let i=0;i<8000;i++){const x=b.minX+Math.random()*(b.maxX-b.minX),y=b.minY+Math.random()*(b.maxY-b.minY);if(safeAt(x,y,poly,r))return[x,y]}return[0,0]}
function scanIntersections(poly,y){const xs=[];for(let i=0,j=poly.length-1;i<p.length;j=i++){const[x1,y1]=poly[j],[x2,y2]=poly[i];if((y1<=y&&y2>y)||(y2<=y&&y1>y))xs.push(x1+(y-y1)*(x2-x1)/(y2-y1))}xs.sort((a,b)=>a-b);const out=[];for(let i=0;i+1<xs.length;i+=2)out.push([xs[i],xs[i+1]]);return out}

function makeLawnTexture(poly,N=210){
  const b=bounds(poly),w=b.maxX-b.minX,h=b.maxY-b.minY,H=Math.max(60,Math.round(N*h/w)),W=N;
  const canvas=document.createElement('canvas');canvas.width=W;canvas.height=H;const ctx=canvas.getContext('2d');
  const inside=new Uint8Array(W*H),cut=new Uint8Array(W*H);let insideCount=0;
  for(let iy=0;iy<H;iy++)for(let ix=0;ix<W;ix++){const x=b.minX+(ix+.5)/W*w,y=b.minY+(iy+.5)/H*h,k=iy*W+ix;if(pointInPoly(x,y,poly)){inside[k]=1;insideCount++}}
  const tex=new THREE.CanvasTexture(canvas);tex.colorSpace=THREE.SRGBColorSpace;tex.magFilter=THREE.LinearFilter;
  function draw(){
    const im=ctx.createImageData(W,H),d=im.data;
    for(let k=0;k<W*H;k++){const q=4*k;if(!inside[k]){d[q+3]=0;continue}
      const noise=((k*1103515245+12345)>>>16)&7;
      if(cut[k]){d[q]=91+noise;d[q+1]=151+noise;d[q+2]=73+noise}
      else{d[q]=37+noise;d[q+1]=103+noise;d[q+2]=47+noise}
      d[q+3]=255}
    ctx.putImageData(im,0,0);tex.needsUpdate=true
  }
  function sampleUncut(x,y,r){
    const cx=(x-b.minX)/w*W,cy=(y-b.minY)/h*H,rx=r/w*W,ry=r/h*H;
    let total=0,uncut=0;
    for(let iy=Math.max(0,Math.floor(cy-ry));iy<=Math.min(H-1,Math.ceil(cy+ry));iy++)
      for(let ix=Math.max(0,Math.floor(cx-rx));ix<=Math.min(W-1,Math.ceil(cx+rx));ix++){
        const dx=(ix+.5-cx)/rx,dy=(iy+.5-cy)/ry,k=iy*W+ix;
        if(dx*dx+dy*dy<=1&&inside[k]){total++;if(!cut[k])uncut++}
      }
    return total?uncut/total:0;
  }
  function mark(x,y,r){const cx=(x-b.minX)/w*W,cy=(y-b.minY)/h*H,rx=r/w*W,ry=r/h*H;let add=0;
    for(let iy=Math.max(0,Math.floor(cy-ry));iy<=Math.min(H-1,Math.ceil(cy+ry));iy++)
      for(let ix=Math.max(0,Math.floor(cx-rx));ix<=Math.min(W-1,Math.ceil(cx+rx));ix++){const dx=(ix+.5-cx)/rx,dy=(iy+.5-cy)/ry,k=iy*W+ix;if(dx*dx+dy*dy<=1&&inside[k]&&!cut[k]){cut[k]=1;add++}}
    return add}
  draw(); return{b,w,h,W,H,inside,cut,insideCount,tex,mark,sampleUncut,draw}
}

function dockPoint(poly){
  const b=bounds(poly),m=MOWER_CLEARANCE+.14;
  const tries=[[b.minX+m,0],[b.minX+m,b.minY+m],[0,b.minY+m],[b.maxX-m,0]];
  for(const p of tries)if(safeAt(p[0],p[1],poly,MOWER_CLEARANCE))return p;
  return randomPoint(poly,MOWER_CLEARANCE)
}

function straightInside(a,b,poly,r=MOWER_CLEARANCE){
  const d=Math.hypot(b[0]-a[0],b[1]-a[1]);
  const n=Math.max(1,Math.ceil(d/.055));
  for(let i=0;i<=n;i++){
    const t=i/n,x=a[0]*(1-t)+b[0]*t,y=a[1]*(1-t)+b[1]*t;
    if(!safeAt(x,y,poly,r))return false;
  }
  return true;
}

function bfsPath(start,goal,poly,step=.28,r=MOWER_CLEARANCE){
  if(!safeAt(start[0],start[1],poly,r) || !safeAt(goal[0],goal[1],poly,r))return null;
  if(straightInside(start,goal,poly,r))return[goal];

  const b=bounds(poly);
  const nx=Math.ceil((b.maxX-b.minX)/step)+1,ny=Math.ceil((b.maxY-b.minY)/step)+1;
  const id=(x,y)=>y*nx+x,coord=i=>[i%nx,Math.floor(i/nx)];
  const world=(x,y)=>[b.minX+x*step,b.minY+y*step];
  const cell=p=>[
    Math.max(0,Math.min(nx-1,Math.round((p[0]-b.minX)/step))),
    Math.max(0,Math.min(ny-1,Math.round((p[1]-b.minY)/step)))
  ];

  // Find a nearby safe grid node for start/goal instead of ever allowing
  // an unsafe direct fallback.
  function nearestSafe(c,p){
    let best=null,bestD=Infinity;
    for(let rad=0;rad<=5 && !best;rad++){
      for(let yy=c[1]-rad;yy<=c[1]+rad;yy++)for(let xx=c[0]-rad;xx<=c[0]+rad;xx++){
        if(xx<0||yy<0||xx>=nx||yy>=ny)continue;
        const q=world(xx,yy);
        if(!safeAt(q[0],q[1],poly,r))continue;
        if(!straightInside(p,q,poly,r))continue;
        const d=Math.hypot(q[0]-p[0],q[1]-p[1]);
        if(d<bestD){best=[xx,yy];bestD=d}
      }
    }
    return best;
  }

  const sc=nearestSafe(cell(start),start),tc=nearestSafe(cell(goal),goal);
  if(!sc||!tc)return null;

  const q=[id(...sc)],prev=new Int32Array(nx*ny);prev.fill(-2);prev[q[0]]=-1;
  let qi=0,endId=-1;
